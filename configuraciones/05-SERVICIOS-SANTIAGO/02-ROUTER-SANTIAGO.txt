
! ==========================================
! CONFIGURACION SANTIAGO - SCRIPT COMPLETO
! ==========================================
configure terminal

! ------------------------------------------
! 1. ENCEDER INTERFAZ PRINCIPAL
! ------------------------------------------
interface e0/1
 no shutdown
 exit

! ------------------------------------------
! 2. SUBINTERFACES VLAN (ROUTER-ON-A-STICK)
! ------------------------------------------

! VLAN 25 - Servidores / Data Center
interface e0/1.25
 encapsulation dot1Q 25
 ip address 192.168.4.1 255.255.255.128
 no shutdown
 exit

! VLAN 30 - Ventas (DHCP Relay apuntando al Server)
interface e0/1.30
 encapsulation dot1Q 30
 ip address 192.168.5.1 255.255.255.128
 ip helper-address 192.168.4.10
 no shutdown
 exit

! VLAN 35 - Administracion (DHCP Relay apuntando al Server)
interface e0/1.35
 encapsulation dot1Q 35
 ip address 192.168.5.129 255.255.255.128
 ip helper-address 192.168.4.10
 no shutdown
 exit

! ------------------------------------------
! 3. INTERFAZ WAN (CONEXION ISP / NUBE)
! ------------------------------------------
interface e0/0
 ip address 19.0.0.5 255.255.255.252
 no shutdown
 exit

! ------------------------------------------
! 4. ENRUTAMIENTO ESTÁTICO (RUTA POR DEFECTO)
! ------------------------------------------
ip route 0.0.0.0 0.0.0.0 19.0.0.6

! ------------------------------------------
! 5. GUARDAR CONFIGURACION
! ------------------------------------------
end
write memory

! ==========================================
! 6. COMANDOS DE VERIFICACION
! ==========================================
show ip interface brief
show vlans
show ip route
show ip interface e0/1.30
